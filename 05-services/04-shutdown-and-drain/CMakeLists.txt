# -----------------------------------------------------------------------------
# 05-services/04-shutdown-and-drain — SIGTERM, then a shutdown that keeps its promises
#
# A single-binary project, so it takes no ROLE and the slug comes from the project
# directory alone:
#
#     05-services/04-shutdown-and-drain  ->  qb-example-services-shutdown-and-drain
#
# It is one program and not a server/client pair on purpose: the subject is a SEQUENCE
# (stop accepting -> drain -> flush -> exit) and a sequence has to be watched from end to
# end. The load is generated by client ACTORS inside the same engine, and the program
# raises SIGTERM against itself at the one moment where the work is genuinely in flight —
# so it demonstrates itself rather than waiting for somebody with a terminal.
# -----------------------------------------------------------------------------

project(qb_examples_05_shutdown_and_drain
        LANGUAGES CXX
        VERSION ${QB_FRAMEWORK_VERSION}
        DESCRIPTION "Graceful shutdown — stop accepting, drain in flight, flush, exit"
)

qb_status_message("Building qb examples — 05-services/04-shutdown-and-drain")

# `main.cpp` first: for a multi-file PROJECT the target's slug comes from the project directory, so
# the source order is free — but `dev/agent/run-examples.py` records the FIRST source as the one
# whose header block carries the `@expect` lines, and the header block lives with `main()`.
qb_example(
        SOURCE
        main.cpp
        load.cpp
        DEPENDS qb-core
)
